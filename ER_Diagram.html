Hold Ctrl and scroll your mouse wheel to zoom in/out. Scroll horizontally/vertically to pan.

erDiagram
    Users {
        int user_id PK
        varchar username "UNIQUE"
        varchar password "SHA-256"
        varchar full_name
        varchar role "Admin/Reception"
        timestamp created_at
    }

    Specializations {
        int specialization_id PK
        varchar specialization_name
    }

    Patients {
        int patient_id PK
        varchar patient_name
        varchar gender
        int age
        varchar blood_group
        varchar contact
        varchar address
        date registration_date
    }

    Doctor {
        int doctor_id PK
        varchar doctor_name
        int specialization_id FK
        varchar qualification
        decimal consultation_fee
        varchar contact
    }

    Doctor_Schedule {
        int schedule_id PK
        int doctor_id FK
        varchar day_of_week
        time start_time
        time end_time
    }

    Appointments {
        int appointment_id PK
        int patient_id FK
        int doctor_id FK
        date appointment_date
        time appointment_time
        varchar room_number
        varchar status
    }

    Prescriptions {
        int prescription_id PK
        int appointment_id FK
        varchar diagnosis
        varchar medicine
        date next_visit_date
        varchar remarks
    }

    Billing {
        int bill_id PK
        int appointment_id FK
        decimal amount
        date bill_date
        varchar payment_method
        varchar payment_status
    }

    Feedback {
        int feedback_id PK
        int patient_id FK
        int rating
        date feedback_date
        varchar comments
    }

    Emergency {
        int emergency_id PK
        int patient_id FK
        varchar emergency_type
        varchar priority_level
        varchar status
        int assigned_doctor FK
        date arrival_date
        time arrival_time
    }

    Specializations ||--o{ Doctor : "has"
    Doctor ||--o{ Doctor_Schedule : "has schedule"
    
    Patients ||--o{ Appointments : "books"
    Doctor ||--o{ Appointments : "attends"
    
    Appointments ||--o| Prescriptions : "results in"
    Appointments ||--o| Billing : "generates"
    
    Patients ||--o{ Feedback : "provides"
    
    Patients ||--o{ Emergency : "admitted as"
    Doctor ||--o{ Emergency : "assigned to"
